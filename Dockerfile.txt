FROM python:3.10-slim

WORKDIR /app

# Copiamos el archivo de dependencias y las instalamos
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copiar los scripts al contenedor
COPY consumidor_v2x.py .
COPY productor_v2x.py .

# Exponer los puertos para SNS y MQTT
EXPOSE 80
EXPOSE 1883

# Ejecutar el script principal al arrancar
CMD ["python3", "-u", "consumidor_v2x.py"]
